cmake_minimum_required(VERSION 3.13)
project(universr_axera)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -mcpu=cortex-a55 -mtune=cortex-a55 -ffast-math -funroll-loops -fopenmp -fvisibility=hidden")
if(NOT AX650)
 message(FATAL_ERROR "Use -DAX650=ON")
endif()
if(NOT BSP_MSP_DIR)
 set(BSP_MSP_DIR ${CMAKE_SOURCE_DIR}/../toolchains/ax650n_bsp_sdk/msp/out)
endif()
include_directories(${BSP_MSP_DIR}/include ${CMAKE_SOURCE_DIR})
link_directories(${BSP_MSP_DIR}/lib)
add_executable(universr_ax650 main.cpp third_party/kissfft/kiss_fft.c third_party/kissfft/kiss_fftr.c)
target_link_libraries(universr_ax650 ax_sys ax_engine ax_interpreter pthread m gomp)
install(TARGETS universr_ax650 RUNTIME DESTINATION ./)
